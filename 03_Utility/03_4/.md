// 1. #forward-btnというIDを持つbutton要素を取得
// 2. #back-btnというIDを持つbutton要素を取得
// 3. #slider imgの複数要素を取得

// 8-2. 5の変数をデクリメントして0を下回ったら => a

<details>
<summary>解答例</summary>

```JS
const forwardBtn = document.querySelector("#forward-btn");
const backBtn = document.querySelector("#back-btn");
const slideImages = document.querySelectorAll("#slider img");
const indexLabel = document.querySelector("#index-label");
let currentIndex = 0;

document.addEventListener("DOMContentLoaded", () => {
    // 初回のみ最初の要素を表示する
    setVisibility(0, 0);
});

forwardBtn.addEventListener("click", () => {
    const lastIndex = currentIndex;
    if (++currentIndex >= slideImages.length) {
        currentIndex = 0;
    }
    setVisibility(lastIndex, currentIndex);
});

backBtn.addEventListener("click", () => {
    const lastIndex = currentIndex;
    if (--currentIndex < 0) {
        currentIndex = slideImages.length - 1;
    }
    setVisibility(lastIndex, currentIndex);
});

function setVisibility(last, now) {
    slideImages[last].classList.remove("visible");
    slideImages[now].classList.add("visible");
    indexLabel.textContent = `${currentIndex + 1}/${slideImages.length}`;
}
```
